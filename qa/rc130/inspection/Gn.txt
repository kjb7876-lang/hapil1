      e.fillRect(-13, -23, 26, 7));
  } else if (n.kind === `time`) {
    let t = n.size * (8 + a * 8);
    (e.rotate(-a * Math.PI * 1.5),
      (e.strokeStyle = n.color),
      (e.lineWidth = n.ultimate ? 5 : 3),
      e.beginPath(),
      e.ellipse(0, 0, t, t * 0.48, 0, 0, Math.PI * 2),
      e.stroke());
    let r = i.lowFx ? 8 : 16;
    for (let n = 0; n < r; n++) {
      let i = (n / r) * Math.PI * 2;
      (e.beginPath(),
        e.moveTo(Math.cos(i) * t * 0.8, Math.sin(i) * t * 0.38),
        e.lineTo(Math.cos(i) * t, Math.sin(i) * t * 0.48),
        e.stroke());
    }
    (e.beginPath(),
      e.moveTo(0, 0),
      e.lineTo(Math.cos(a * 5) * t * 0.55, Math.sin(a * 5) * t * 0.27),
      e.moveTo(0, 0),
      e.lineTo(Math.cos(-a * 2.1) * t * 0.35, Math.sin(-a * 2.1) * t * 0.17),
      e.stroke());
  } else if (n.kind === `cross`) {
    let t = n.size * (8 + a * 7);
    ((e.strokeStyle = n.color),
      (e.lineWidth = n.ultimate ? 7 : 4),
      e.rotate(a * Math.PI * 0.65),
      e.beginPath(),
      e.moveTo(-t, 0),
      e.lineTo(t, 0),
      e.moveTo(0, -t * 0.7),
      e.lineTo(0, t * 0.7),
      e.stroke(),
      (e.strokeStyle = n.accent),
      (e.lineWidth *= 0.35),
      e.strokeRect(-t * 0.44, -t * 0.24, t * 0.88, t * 0.48));
  } else if (n.kind === `petals`)
    for (let t = 0; t < m + 5; t++) {
      let r = (t / (m + 5)) * Math.PI * 2 + a * (t % 2 ? 4 : -4),
        i = a * n.size * (10 + (t % 4) * 3);
      (e.save(),
        e.translate(Math.cos(r) * i, Math.sin(r) * i * 0.5 - a * 12),
        e.rotate(r + a * 4),
        (e.fillStyle = t % 3 ? n.color : n.accent),
        e.beginPath(),
        e.ellipse(0, 0, 8, 2.5, 0, 0, Math.PI * 2),
        e.fill(),
        e.restore());
    }
  else if (n.kind === `glitch`) {
    let t = i.lowFx ? 7 : 16;
    for (let r = 0; r < t; r++) {
      let t = Math.sin((n.id + r * 17) * 91.13) * Math.PI * 2 + a * 3,
        i = n.size * (5 + (r % 5) * 4) * a,
        o = 5 + (r % 4) * 5;
      ((e.fillStyle = r % 3 ? n.color : n.accent),
        e.fillRect(
          Math.cos(t) * i - o / 2,
          Math.sin(t) * i * 0.48 - 2,
          o,
          4 + (r % 3),
        ));
    }
    ((e.strokeStyle = n.accent),
      (e.lineWidth = 2),
      e.strokeRect(-n.size * 8, -n.size * 7, n.size * 16, n.size * 11));
  } else if (n.kind === `mark`) {
    let t = n.size * (7 + a * 6);
    (e.rotate(a * Math.PI),
      (e.strokeStyle = n.color),
      (e.lineWidth = n.ultimate ? 5 : 3),
      e.beginPath());
    for (let n = 0; n < 3; n++) {
      let r = -Math.PI / 2 + (n * Math.PI * 2) / 3,
        i = Math.cos(r) * t,
        a = Math.sin(r) * t * 0.55;
      n ? e.lineTo(i, a) : e.moveTo(i, a);
    }
    (e.closePath(),
      e.stroke(),
      e.rotate(-a * Math.PI * 2),
      e.strokeRect(-t * 0.45, -t * 0.25, t * 0.9, t * 0.5));
  } else {
    let t = n.size * (7 + a * 12);
    ((e.strokeStyle = n.color),
      (e.lineWidth = n.ultimate ? 6 : 3),
      e.beginPath(),
      e.ellipse(0, 0, t, t * 0.46, 0, 0, Math.PI * 2),
      e.stroke());
    for (let r = 0; r < m; r++) {
      let i = (r / m) * Math.PI * 2 + n.id;
      ((e.strokeStyle = r % 2 ? n.color : n.accent),
        e.beginPath(),
        e.moveTo(Math.cos(i) * t * 0.35, Math.sin(i) * t * 0.16),
        e.lineTo(Math.cos(i) * t, Math.sin(i) * t * 0.46),
        e.stroke());
    }
  }
  e.restore();
}
